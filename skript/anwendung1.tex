
Die Homotopietheorie beschäftigt sich unter anderem mit der Frage,
wann zwei Räume homotopieäquivalent sind.
Diese Frage lässt sich in interessanten Fällen kaum direkt beantworten,
sondern nur über den Umweg von Invarianten --- also Mathematischen Objekten,
die man Räumen so zuordnet, dass diese Objekte für homotopieäquivalente Räume stets gleich sind.
Wichtige Invarianten sind \emph{Homotopiegruppen} und \emph{Kohomologiegruppen}.

Klassisch beschäftigt sich die Homotopietheorie mit (speziellen) topologischen Räumen,
aber ein Großteil der Arbeit kann auf Seite der \emph{Homotopietypen} erledigt werden,
zu der wir durch die Homo\-to\-pie\-ty\-pen\-theo\-rie Zugang haben.

\subsection{Homotopiegruppen}

\begin{definition}
  Sei \(A\) punktierter Typ und \(n:\N\).
  Dann ist
  \[\pi_n\colonequiv \|\Omega^n(A)\|_0\]
  --- was wir für \(n\geq 1\) die \begriff{\(n\)-te Homotopiegruppe}\index{\(\pi_n(A)\)} von \(A\) nennen.
\end{definition}

Um den Namen zu rechtfertigen, sollten wir zeigen, dass es sich bei den \(\pi_n(A)\) um Gruppen handelt.
Tatsächlich ist \(\pi_n(A)\) für \(n\geq 2\) eine abelsche Gruppe, was wir zunächst als stärkeres Resultat für doppelte Schleifenräume zeigen und als Eckmann-Hilton Argument bekannt ist:

\begin{lemma}
  \label{eckmann-hilton}
  Sei \(A\) punktiert. Für \(x,y:\Omega^2(A)\) haben wir \(x\kon y=y\kon x\).
\end{lemma}

\begin{beweis}
  Wir betrachten die folgenden Situation:
  \begin{center}
    \begin{tikzcd}[nodes={row sep=1.4cm, column sep=1.6cm}]
      x\ar[r,bend left=50,swap,""{name=UL, below},"p"{above}]\ar[r,bend right=50,swap,""{name=DL, above},"q"{below}] & y\ar[r,bend left=50,swap,""{name=UR, below},"s"{above}]\ar[r,bend right=50,swap,""{name=DR, above},"t"{below}] & z \\
      \arrow[Rightarrow, from=UL, to=DL, "\alpha"]
      \arrow[Rightarrow, from=UR, to=DR, "\beta"]
    \end{tikzcd}
  \end{center}
  Wir wollen eine Operation \(\ast\) definieren, die aus diesem Datum eine vertikal konkatenierte Gleichheit von Gleichheiten macht, also etwas dieser Gestalt:
  \begin{center}
    \begin{tikzcd}[nodes={row sep=1.4cm, column sep=1.6cm}]
      x\ar[r,bend left=50,swap,""{name=UL, below},"p\kon s"{above}]\ar[r,bend right=50,swap,""{name=DL, above},"q\kon t"{below}] & z \\
      \arrow[Rightarrow, from=UL, to=DL, "\alpha\ast\beta"]
    \end{tikzcd}
  \end{center}
  Dazu definieren wir zwei Operation, die eine Gleichheit links bzw.\ rechts an eine Gleichheit von Gleichheiten anhängt, mithilfe zweier Zeugen \(\mathrm{ru}(p):p\kon\refl=p\) und \(\mathrm{lu}(p):\refl\kon p=p\):
  \begin{align*}
    \_\kon_l\_ &: (p:x=y)\to s=t \to p\kon s=p\kon t \\
    \refl\kon_l \beta&\colonequiv \mathrm{lu}(s)\kon\beta \kon\mathrm{lu}(t)^{-1}
  \end{align*}
  und 
  \begin{align*}
    \_\kon_r\_ &: p=q\to (s:y=z) \to p\kon s=q\kon s \\
    \alpha\kon_r \refl&\colonequiv \mathrm{ru}(p)\kon\alpha \kon\mathrm{ru}(q)^{-1}
  \end{align*}
  Dann bekommen wir zwei Operation
  \begin{align*}
    \alpha\ast\beta&\colonequiv (\alpha\kon_r s) \kon (q\kon_l\beta) \\
    \alpha\ast^\prime\beta&\colonequiv (p\kon_l\beta)\kon (\alpha\kon_r t)
  \end{align*}
\end{beweis}

\begin{bemerkung}
  Sei \(A\) punktierter Typ.
  \begin{enumerate}[(a)]
  \item Für \(n\geq 1\) ist \(\pi_n(A)\) eine Gruppe mit der durch Konkatenation von Gleichheiten gegebenen Verknüpfung.
  \item Für \(n\geq 2\) ist \(\pi_n(A)\) eine abelsche Gruppe\footnote{D.h.\ \(x\kon y=y\kon x\) für alle \(x,y\).}.
  \end{enumerate}
\end{bemerkung}

\begin{beweis}
  \begin{enumerate}[(a)]
  \item Wir haben die entsprechenden Gesetze bereits für allgemeine Identitätstypen \(x=y\) gezeigt.
    Nun ist \(\|\Omega^n(A)\|_0\) eine Menge für \(x,y,z:|\Omega^n(A)\|_0\) gilt zum Beispiel Assoziativität per \(\|\_\|_0\)-Induktion und Andwenden von \(|\_|_0\) auf die entsprechende Gleichheit in \(\Omega^n(A)\).
  \item \Cref{eckmann-hilton}.
  \end{enumerate}
\end{beweis}

\begin{beispiel}
  \begin{enumerate}[(a)]
  \item \(\pi_1(S^1)=\Z\)
  \item \(\pi_1(\BAut_{\mU}(\zwei))=\zwei\)
  \end{enumerate}
\end{beispiel}

\begin{definition}
  Ein Typ \(A\) heißt \begriff{\(n\)-zusammenhängend}, wenn \(\|A\|_n\) kontrahierbar ist.
\end{definition}

\begin{bemerkung}
  \begin{enumerate}[(a)]
  \item Ein punktierter Typ \(A\) ist genau dann zusammenhängend, wenn \(A\) \(0\)-zusammenhängend ist.
  \item Wenn ein Typ \(A\) \(n\)-zusammenhängend ist, dann ist \(A\) auch \(k\)-zusammenhängend, für alle \(k\leq n\).
  \end{enumerate}
\end{bemerkung}

\begin{beweis}
  \begin{enumerate}[(a)]
  \item \(((x:A)\to \|x=\ast\|) = (y:\|A\|_0)\to |y|_0=|\ast|\),
  wobei die rechte Seite bedeutet, dass \(\|A\|_0\) kontrahierbar ist.
\item Für jeden \(k\)-Typ \(B\) gilt
  \[(\|\|A\|_n\|_k\to B)\simeq (\|A\|_n\to B)\simeq (A\to B)\]
  wobei die Komposition dieser Äquivalenzen durch \(\_\circ (|\_|_k \circ |\_|_n)\) gegeben ist.
  Mit \Cref{n-abschneidung-berechnen} ist also \(\|\|A\|_n\|_k\simeq \|A\|_k\) und damit \(\|A\|_k=\eins\), wenn \(\|A\|_n=\eins\).
  \end{enumerate}
\end{beweis}

\begin{lemma}
  Sei \(A\) punktierter Typ.
  \begin{enumerate}[(a)]
  \item \(\Omega\|A\|_{n+1}=\|\Omega(A)\|_n\)
  \item \(\pi_n(A)=\|\Omega^nA\|_0=\Omega^n\|A\|_n\)
  \end{enumerate}
\end{lemma}

\begin{beweis}
  \begin{enumerate}[(a)]
  \item Mit \Cref{n-abschneidung-gleichheit}.
  \item (a) und Induktion.
  \end{enumerate}
\end{beweis}

\begin{lemma}
  \begin{enumerate}[(a)]
  \item Ist \(A\) ein \(n\)-Typ, dann gilt \(\pi_k(A)=\{0\}\) für \(k>n\).
  \item Ist \(A\) ein \(n\)-zusammenhängender Typ, so gilt \(\pi_k(A)=\{0\}\) für \(k\leq n\).
  \end{enumerate}
\end{lemma}

\begin{theorem}
  Sei \(A\) \(n\)-zusammenhängend, dann ist \(\Sigma A\) \((n+1)\)-zusammenhängend.
\end{theorem}

\begin{beweis}
  Für jeden \((n+1)\)-Typ \(B\) gilt
  \[(\|\Sigma A\|_{n+1}\to B)\simeq (\Sigma A\to B) \simeq \left(\sum_{b_N,b_S:B}A\to b_N=b_S\right)\simeq \left(\sum_{b_N,b_S:B}b_N=b_S\right)\simeq B\]
\end{beweis}

\begin{korollar}
  \begin{enumerate}
  \item \(S^{n+1}\) ist \(n\)-zusammenhängend.
  \item \(\pi_k(S^n)=\{0\}\) für \(k<n\).
  \end{enumerate}
\end{korollar}

\begin{definition}
  Sei eine punktierte Abbildung \(f:A\to_\ast B\), also \(f:A\to B\) mit \({p:f(\ast)=\ast}\) gegeben.
  \begin{enumerate}[(a)]
  \item Es gibt eine induzierte Abbildung
    \begin{align*}
      \Omega f : \Omega A &\to \Omega B \\
      (l:\ast=_A\ast)&\mapsto p^{-1}\kon l\kon p
    \end{align*}
  \item Induktiv ist damit auch 
    \begin{align*}
      \Omega^{n+1} f &: \Omega^{n+1} A \to \Omega^{n+1} B \\
      \Omega^{n+1} f&\colonequiv \Omega(\Omega^{n}(f))
    \end{align*}
    gegeben.
  \item \(\pi_n(f)\colonequiv \|\Omega^n(f)\|_0\).
  \end{enumerate}
\end{definition}

\begin{lemma}
  \begin{enumerate}[(a)]
  \item \(\pi_n(\id)=\id\)
  \item \(\pi_n(f\circ g)=\pi_n(f)\circ\pi_n(g)\)
  \item \(\pi_n(f)\) ist für \(n\geq 1\) Gruppenhomomorphismus.
  \item \(\pi_n\) bildet Äquivalenzen auf Äquivalenzen ab.
  \end{enumerate}
\end{lemma}

\begin{beweis}
  Für (c) berechnen wir zuerst:
  \(\Omega f(s\kon t)=p^{-1}\kon s\kon t\kon p=p^{-1}\kon s\kon p\kon p^{-1}\kon t\kon p=\Omega f(s)\kon\Omega f(t)\)
  Das überträgt sich auf \(\Omega^n(f)\) und \(\pi_n(f)\).
\end{beweis}

\begin{definition}
  Eine Sequenz
  \begin{center}
    \begin{tikzcd}
      \dots\ar[r,"f_{n-1}"] & A_n\ar[r,"f_n"] & A_{n+1}\ar[r,"f_{n+1}"] &\dots \\
    \end{tikzcd}
  \end{center}
  von punktierten Typen und Abbildungen heißt \begriff{exakt} bei \(A_{n+1}\),
  wenn \[\im(f_{n})=\ker({f_{n+1}})\colonequiv ((x:A_n)\times f_{n}(x)=\ast)=\fib_{f_n}(\ast)\]
  gilt. Die Sequenz heißt exakt, wenn sie an jeder Stelle exakt ist.
  Sind nicht für alle \(n:\N\) Abbildungen gegeben, so heißt eine Sequenz exakt, sobald sie an jeder Stelle mit komponierbaren Abbildungen exakt ist.
\end{definition}

\begin{lemma}
  Sei
  \begin{center}
    \begin{tikzcd}
    A\ar[r] & B\ar[r,"f"] & C\ar[r] & D   
    \end{tikzcd}
  \end{center}
  eine exakte Sequenz von Gruppenhomomorphismen.
  \begin{enumerate}[(a)]
  \item Wenn \(A=\{0\}\) gilt, dann ist \(f\) eine Einbettung.
  \item Wenn \(D=\{0\}\) gilt, dann ist \(f\) surjektiv.
  \item Wenn \(A=\{0\}\) und \(D=\{0\}\) gilt, dann ist \(f\) ein Isomorphismus.
  \end{enumerate}
\end{lemma}

\begin{lemma}
  \label{0-abschneidung-faser-exakt}
  Für eine punktierte Abbildung \(f:A\to_\ast B\) ist die folgende Sequenz exakt:
  \begin{center}
    \begin{tikzcd}
    \|\fib_f(\ast)\|_0\ar[r,"\|\pi_1\|_0"] & \|A\|_0\ar[r,"\|f\|_0"] & \|B\|_0  
    \end{tikzcd}
  \end{center}
\end{lemma}


\begin{beweis}
  Sei \(x:\|A\|_0\) mit \(\|f\|_0(f)=\ast\) gegeben.
  Per Induktion ist \(y:A\) mit \(p:|f(\ast)|=|\ast|\) bzw \(q:\|f(\ast)=\ast\|\).
  Daraus bekommen wir ein Element der Faser, da wir eine Aussage zeigen wollen.
\end{beweis}

\begin{definition}
  Eine Sequenz von punktierten Abbildungen
  \begin{center}
    \begin{tikzcd}
      \dots\ar[r,"f_{n-1}"] & A_n\ar[r,"f_n"] & A_{n+1}\ar[r,"f_{n+1}"] & \dots 
    \end{tikzcd}
  \end{center}
  heißt \begriff{Fasersequenz}, wenn für jedes \(n:\Z\), \(f_{n-1}\) bis auf Äquivalenz die Faser von \(f_n\) ist, es also eine punktierte Äquivalenz \(\fib_{f_n}(\ast)\simeq A_{n-1}\) gibt, sodass
  \begin{center}
    \begin{tikzcd}
      \fib_{f_n}(\ast)\ar[d,"\simeq",swap]\ar[r,"\pi_1"] & A_n \\
      A_{n-1}\ar[ru,"f_{n-1}",swap]
    \end{tikzcd}
  \end{center}
  (als Diagramm punktierter Abbildungen) kommutiert.
\end{definition}

\begin{lemma}
  \label{fasersequenz}
  Für eine punktierte Abbildung \(f:A\to_\ast B\) gibt es die durch das folgende Diagramm beschriebenen Äquivalenzen und Homotopien:
  \begin{center}
    \begin{tikzcd}
 & \fib_{p_1}(\ast)\ar[r,"p_2"]\ar[d,"\simeq"] & \fib_f(\ast)\ar[r,"p_1"] & A\ar[r,"f"] & B \\
 \fib_{v}(\ast)\ar[r,"p_3"]\ar[d,"\simeq"] & \Omega B\ar[ru,swap,"v"] &  & & \\
 \Omega A\ar[ru,"\Omega f\circ(\_)^{-1}",swap]
    \end{tikzcd}
  \end{center}
  wobei \(p_i\) die jeweilige erste Projektion der Faser ist. Durch Iteration ergibt sich also die folgende Fasersequenz:
  \begin{center}
    \begin{tikzcd}
      & & \dots\ar[lld] \\
      \Omega^2 \fib_f(\ast)\ar[r,"\Omega^2 p_1"] & \Omega A\ar[r,"\Omega^2 f"] & \Omega^2 B\ar[lld] \\
      \Omega \fib_f(\ast)\ar[r,"\Omega p_1\circ(\_)^{-1}"] & \Omega A\ar[r,"\Omega f\circ(\_)^{-1}"] & \Omega B\ar[lld] \\
      \fib_f(\ast)\ar[r,"p_1"] & A\ar[r,"f"] & B 
    \end{tikzcd}
  \end{center}
\end{lemma}

\begin{beweis}
  \begin{align*}
    \fib_{p_1}(\ast)&\equiv \sum_{(x,p):\fib_f(\ast)}p_1(x,p)=\ast \\
    &\equiv \sum_{(x,p):\fib_f(\ast)}x=\ast \\
    &\simeq (x:A)\times (x=\ast)\times (p:f(x)=\ast) \\
    &\simeq f(\ast)=\ast \\
    &\simeq \ast=\ast \\
    &\equiv \Omega B 
  \end{align*}
  Wobei die letzte Äquivalenz \(q^{-1}\kon\_\) ist, für das durch die Punktiertheit von \(f\) gegebene \(q:f(\ast)=\ast\).
  Eine mit der Projektion kommutierende Abbildung \(\Omega B\to \fib_f(\ast)\) ist also durch
  \[v\colonequiv (l:\Omega B)\mapsto (\ast,q\kon l) \]
  gegeben.
\end{beweis}

\begin{theorem}
  \label{les}
  Für jede punktierte Abbildung gibt es eine lange exakte Sequenz:
  \begin{center}
    \begin{tikzcd}
      & & \dots\ar[lld] \\
      \pi_2(\fib_f(\ast))\ar[r,"\pi_2(p_1)"] & \pi_2 A\ar[r,"\pi_2(f)"] & \pi_2(B)\ar[lld] \\
      \pi_1( \fib_f(\ast))\ar[r,"\pi_1(p_1)\circ(\_)^{-1}"] & \pi_1(A)\ar[r,"\pi_1(f)\circ(\_)^{-1}"] & \pi_1(B)\ar[lld] \\
      \pi_0(\fib_f(\ast))\ar[r,"\pi_0(p_1)"] & \pi_0(A)\ar[r,"\pi_0(f)"] & \pi_0(B)
    \end{tikzcd}
  \end{center}
\end{theorem}

\begin{beweis}
  Anwenden von \(\|\_\|_0\) auf die Fasersequenz aus \Cref{fasersequenz} und \Cref{0-abschneidung-faser-exakt}.
\end{beweis}

\begin{bemerkung}
  Das Vorzeichen in \Cref{les} lässt sich auch entfernen, ist aber für Funktorialität der Konstruktion nötig.
\end{bemerkung}

Bevor wir uns diesen im nächsten Abschnitt zuwenden, werden wir aber noch \emph{Pullbacks}, auch \emph{Faserprodukt} genannt anschauen.
Diese Konstruktion aus der Kategorientheorie ist uns bereits in vielen Formen begegnet und liefert noch eine weitere Perspektive auf den Beweis von \Cref{fasersequenz}.

\begin{definition}
Seien \(A,B,C:\mU\) und \(f:A\to C\), \(g:B\to C\).
Ein Datum 
\begin{tikzcd}
 A\ar[r,"f"] & C & B\ar[l,"g",swap]
\end{tikzcd}
heißt \begriff{Kospann}. Der Typ
\[
  \mathrm{PB}(f,g)\colonequiv A\times_C B\colonequiv \sum_{x:A}\sum_{y:B}f(x)=g(y)
\]
heißt (kanonischer) \begriff{Pullback}\index{\(A\times_C B\)} oder \begriff{Faserprodukt} des Kospanns 
\begin{tikzcd}
 A\ar[r,"f"] & C & B\ar[l,"g",swap]
\end{tikzcd}.
\end{definition}

\begin{beispiel}
\begin{enumerate}[(a)]
\item Eine Faser \(\fib_f(y)\) ist bis auf Äquivalenz der Pullback \(\mathrm{PB}(f,(\_:\eins)\mapsto y)\).
\item Der Identitätstyp \(x=_A y\) ist äquivalent zum Pullback \(\mathrm{PB}((\_:\eins)\mapsto x, (\_:\eins)\mapsto y)\).
\end{enumerate}
\end{beispiel}

\subsection{Pushout und Join}
\begin{regeln}
Seien \(A,B,C\) Typen und \(f:C\to A\), \(g:C\to B\).
\begin{itemize}
\item Es gibt den Typ \(A\sqcup_{f,C,g} B\equiv A\sqcup_{C} B\), den \begriff{Pushout}\index{\(A \sqcup_C B\)} des \begriff{Spanns}
\begin{tikzcd}
  A & C\ar[l,"f",swap]\ar[r,"g"] & B 
\end{tikzcd}
\item 
\begin{enumerate}[\ ]
\item Für \(x:A\) gibt es \(\iota_1(x):A\sqcup_C B\), 
\item für \(y:B\) gibt es \(\iota_2(y):A\sqcup_C B\) 
\item und für \(z:C\) gibt es \(\mathrm{glue}(z)\equiv \mathrm{g}(z) : \iota_1(f(z))=\iota_2(g(z))\).
\end{enumerate}
\item Sei \(E:A\sqcup_C B\to \mU\). Für
\begin{enumerate}[\ ]
\item \(e_A:(x:A)\to E(\iota_1(x))\),
\item \(e_B:(y:B)\to E(\iota_2(y))\) und
\item \(e_{\mathrm{g}}:(z:C)\to \transp_E(\mathrm{glue}(z))(e_A(f(z)))=e_B(g(z))\)
\end{enumerate}
gibt es \(\elim_{\sqcup_{\_},E}(e_A,e_B,e_{\mathrm{g}}):(u:A\sqcup_C B)\to E(u)\).
\item \dots sodass gilt
\begin{enumerate}[\ ]
\item \(\elim_{\sqcup_{\_},E}(e_A,e_B,e_{\mathrm{g}})(\iota_1(x))\equiv e_A(x)\),
\item \(\elim_{\sqcup_{\_},E}(e_A,e_B,e_{\mathrm{g}})(\iota_2(y))\equiv e_B(y)\) und
\item \(\apd_{\elim_{\sqcup_{\_},E}(e_A,e_B,e_{\mathrm{g}})}(\mathrm{glue}(z))=e_{\mathrm{g}}(z)\).
\end{enumerate}
\end{itemize}
\end{regeln}

\begin{lemma}
  Sei
\begin{tikzcd}
  A & C\ar[l,"f",swap]\ar[r,"g"] & B 
\end{tikzcd} und für \(Z:\mU\)
\[
  \mathrm{Cocone}(f,g)(Z)\colonequiv \sum_{u:A \to Z}\sum_{v:B\to Z} u\circ f\sim v\circ g\rlap{.}
\]
  Dann ist
  \[
    \varphi\mapsto (\varphi\circ \iota_1,\varphi\circ\iota_2,(x:C)\mapsto \ap_\varphi\mathrm{glue(x)}) : (A\sqcup_C B\to Z)\to \mathrm{Cocone}(f,g)(Z)
  \]
  eine Äquivalenz.
\end{lemma}

\begin{definition}
  Ein \begriff{H-Raum} ist ein punktierter Typ \(A\) zusammen mit \(\mu:A\to A\to A\)
  und Homotopien
  \begin{itemize}
  \item \(\mu(\_,\ast)\sim\id \),
  \item \(\mu(\ast,\_)\sim\id \).
  \end{itemize}
\end{definition}

\begin{beispiel}
  \(S^1\) ist ein H-Raum mit der ``komplexen Multiplikation'', also \(\mu:S^1\to S^1\to S^1\) gegeben durch
  \[\mu(\ast)\colonequiv \id_{S^1}\]
  und \(\mu(l)\coloneq L\), wobei \(L:\id_{S^1}=\id_{S^1}\) bzw \(L:\id_{S^1}\sim \id_{S^1}\) gegeben ist durch
  \begin{align*}
    &L(\ast)\colonequiv l \\
    &L(l)\coloneq \_ : \transp_{\_=\_}(l)(l)=l
  \end{align*}
  wobei \(\_\) ein Zeuge für \(l^{-1}\kon l\kon l=l\) ist. \\
  Wir konstruieren nun noch die beiden Homotopien.
  Es gilt \(\mu(\ast,x)=x\) per Definition. \\
   Die zweite Homotopie, \(\mu(\_,\ast)\sim \id\), konstruieren wir durch Induktion.
   Für \(x\equiv \ast\) haben wir wieder \(\refl:\mu(\ast,\ast)=\ast\) und müssen also noch zeigen:
   \[
      \transp_{\mu(\_,\ast)=\_}(l)(\refl)=\refl
    \]
    was wegen \(\ap_{\mu(\_,\ast)}(l)=l\) \rednote{Begründen} gegeben ist. 
\end{beispiel}

\begin{bemerkung}
  Für das eben definierte \(\mu:S^1\to S^1\to S^1\) ist \(\mu(x)\) stets eine Äquivalenz.
  Es reicht das für \(x\equiv \ast\) zu beweisen und durch die H-Raum-Struktur wissen wir, dass \(\mu(\ast)\) homotop zur Identität ist.
\end{bemerkung}

\begin{definition}
  Die \begriff{Hopf-Faserung} ist der folgende abhängige Typ \(H:S^2\to \mU\),
  gegeben durch (\(\elim_{\Sigma S^1}\)):
  \begin{align*}
    &H(N) \colonequiv S^1 \\
    &H(S) \colonequiv S^1 \\
    &H(m(x)) \coloneq \mathrm{ua}(\mu(x,\_))
  \end{align*}
\end{definition}

\begin{theorem}
  Für den Totalraum der Hopf-Faserung gilt:
  \[
    \sum_{x:S^2}H(x) \simeq S^3
  \]
  und die Faser von \(\pi_1:\sum_{x:S^2}H(x)\to S^2\) ist \(S^1\).
  Damit ist eine Fasersequenz
  \[
     S^1\to S^3\to S^2
   \]
   gegeben.
\end{theorem}

Wir führen nun einen höheren induktiven Typen ein, den Coequalizer, der etwas minimalistischer als der Pushout ist. Es reicht jedoch, zum Beispiel die Kommutativität von abhängigen Summen mit Coequalizern zu zeigen, um diese auch für Pushouts zu erhalten.

\begin{regeln}
  Seien \(A,B\) Typen und \(f,g:A\to B\) Abbildungen.
  \begin{itemize}
  \item Es gibt einen Typ \(\Coeq(f,g)\), den \begriff{Coequalizer} von \(f\) und \(g\).
  \item Es gibt \(\iota:B\to \Coeq(f,g)\) und \(\glue:(a:A)\to \iota(f(a))=\iota(g(a))\).
  \item Sei \(C:\Coeq(f,g)\to \mU\). Für
    \begin{enumerate}[\ ]
    \item \(c_B:(b:B)\to C(b)\) und
    \item \(c_g:(a:A)\to \transp_C(\glue(a))(c_B(f(a)))=c_B(g(a))\)
    \end{enumerate}
    gibt es \(\elim_{\Coeq(f,g)}(c_B,c_g):(x:\Coeq(f,g))\to C(x)\).
  \item \dots sodass gilt
    \begin{enumerate}[\ ]
    \item \(\elim_{\Coeq(f,g)}(c_B,c_g)(\iota(b))\equiv c_B(b)\)
    \item \(\apd_{\elim_{\Coeq(f,g)}(c_B,c_g)}(\glue(a))=c_g(a)\)
    \end{enumerate}
  \end{itemize}
\end{regeln}
