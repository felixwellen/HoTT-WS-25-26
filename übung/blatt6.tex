\blatt

\aufgabe{}
Sei $A:\mU$ und $t\colonequiv \ast\mapsto A: \eins\to \mU$.
Es gilt $\BAut_\mU(A)\simeq \im(t)$.

\aufgabe{}
Seien $A,B:\Prop$ Aussagen und $P:X\to \Prop$ für beliebiges $X:\mU$. Zeige:
\begin{enumerate}[(a)]
\item $A\Rightarrow (\neg\neg A)$
\item $(\neg\neg\neg A) \Rightarrow (\neg A)$
\item $(\neg(A\vee B))\Rightarrow (\neg A) \wedge (\neg B)$
\item $\neg(\forall x:X.P(x))\Rightarrow \neg(\exists x:X.P(x))$
\item $(A\vee \neg A)\Rightarrow ((\neg\neg A) \Rightarrow A)$
\end{enumerate}

\aufgabe{}
  \begin{enumerate}[(a)]
  \item Sei $f:A\to B$. $f$ kann wie folgt faktorisiert werden:
  \begin{center}
    \begin{tikzcd}
      A\ar[dr,"l",swap]\ar[rr,"f"] & & B \\
      & \im(f)\ar[ru,"r",swap] &
    \end{tikzcd}
  \end{center}
  wobei $l$ surjektiv und $r$ eine Einbettung ist.
\item Gegeben sei das folgende Diagramm:
  \begin{center}
    \begin{tikzcd}
      A\ar[r,"t"]\ar[d,"l",swap] & B\ar[d,"r"] \\
      C\ar[r,"b",swap]\ar[ur,dotted,"d"] & D
    \end{tikzcd}
  \end{center}
  Es gelte $rt=bl$ und weiter seien $l$ surjektiv und $r$ eine Einbettung.
  Dann ist der Typ der ``lifts'', also Diagonalen $d:C\to B$ sodass $rd=b$ und $dl=t$ gelten,
  kontrahierbar.
\end{enumerate}

\aufgabe{Äquivalenzinduktion}
  Zeige, dass Äquivalenzinduktion gilt, es also für $P:(A\ B :\mU)\to A\simeq B\to \mU$ und $r_A:P(A,A,\id_A)$ ein
  \[
     \elim_\simeq(r_{\_}):\prod_{A,B:\mU}\prod_{f:A\simeq B}P(A,B,f)\rlap{.}
     \]
     gibt und $\elim_\simeq(r)(A,A,\id_A)=r_A$.
