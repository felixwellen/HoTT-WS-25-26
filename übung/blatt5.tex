\blatt

\aufgabe{}
  Seien $A:\mU$ und $P:A\to\mU$ mit $(x:A)\to \isProp(P(x))$.
  Für $x,y:A$, $u:P(x)$, $v:P(y)$ gilt
  \[
    (x=_A y)\simeq ((x,u)=(y,v))\rlap{.}
  \]

\aufgabe{}
  Seien $A,B$ Typen. Zeige:
  \begin{enumerate}
  \item $\isProp(\isProp(A))$.
  \item $\isProp(\isContr(A))$.
  \item $\isProp(\isNType{n}(A))$.
  \end{enumerate}
  \emph{Hinweis: Lemma 2.3.10 aus dem Skript kann helfen.}

\aufgabe{}
Seien $A:\mU$ und $B:A\to\mU$ sodass $B(x)$ für jedes $x:A$ ein $n$-Typ ist.
Zeige, dass dann auch $(x:A)\to B(x)$ ein $n$-Typ ist.
