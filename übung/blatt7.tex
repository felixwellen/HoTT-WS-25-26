\aufgabe{Lemma von Krauss}
Seien $A:\mU$, $B:\Set$ und $f:A\to B$.
Weiter gebe es $(x,y : A)\to f(x)=f(y)$.
Zeige, dass es dann eine Abbildung $f^\prime:\|A\|\to B$ gibt, sodass das folgende Diagramm kommutiert:
\begin{center}
  \begin{tikzcd}
    A\ar[rr,"f"]\ar[rd,"|\_|",swap] & & B \\
    & \|A\|\ar[ru,"f^\prime",swap] &
  \end{tikzcd}
\end{center}

\aufgabe{Strukturidentitätsprinzip für Monoide}
  Seien \((M,\cdot,e)\) und \((N,\cdot,e)\) Monoide.
  Eine Abbildung \(f : M \to N\) heißt \emph{Homomorphismus}, falls sie ein Homomorphismus von Halbgruppen \((M,\cdot) \to (N,\cdot)\) ist und \(f(e)=e\) erfüllt.
  Ist \(f\) außerdem eine Äquivalenz, so nennen wir \(f\) einen \emph{Isomorphismus}.
  Zeige:
  \[((M,\cdot,e) =_{\mathrm{Mon}} (N,\cdot,e)) \simeq \Iso((M,\cdot,e),(N,\cdot,e))\]

\aufgabe{}
\begin{enumerate}[(a)]
\item Konstruiere $p:\id_{S^1}=\id_{S^1}$, sodass $p\neq \refl_\id$ gilt.
Folgere, dass $S^1\simeq S^1$ keine Menge ist und ein Universum $\mU$, das $S^1$ enthält kein 1-Typ sein kann.
\item Konstruiere eine Abbildung $d:S^1\to S^1$ sodass $\fib_d:S^1\to \mU$ dem abhängigen Typ $D:S^1\to\mU$ aus der Vorlesung entspricht, der durch $D\colonequiv \elim_{S^1,\_\mapsto\mU}(\zwei,\ua(s))$ gegeben ist.
\end{enumerate}

\aufgabe{}
  Sei \(G\) eine Gruppe, \((BG,\ast)\) eine Entschleifung von \(G\) und $t:G\to \Omega(BG)$ ein Isomorphismus.
  Es bezeichne \(G\text{-}\Set\) den Typ der \(G\)-Mengen und wir betrachten die folgende Abbildung:
    \begin{align*}
      d:(BG\to \Set)\to G\text{-}\Set
      d(\rho)\colonequiv (\rho(\ast),g\mapsto \transp_\rho(t(g)))
    \end{align*}
    Sei nun $\rho:BG\to\Set$ und $M\colonequiv \pi_1(d(\rho))$, $\_\cdot\_\colonequiv \pi_2(d(\rho)):G\to M\to M$.
  \begin{enumerate}
  \item  Zeige, dass gilt:
    \[
    \left(\prod_{x:BG}\rho(x)\right) \simeq \sum_{m:M}\prod_{g:G}g\cdot m=m
    \]
  \emph{Hinweis: Aufgabe 1 kann helfen.}
  \item Zeige, dass für $Q\colonequiv (x:BG)\times \rho(x)$ und $m:M$ gilt, dass $(\ast,m)=(\ast,m)$ der Bahn von $m$ entspricht, also äquivalent ist zu $(m^\prime:M)\times \exists g:G. g\cdot m=m^\prime$.
  \end{enumerate}
